src/app/interceptors/token.interceptor.ts

tokenInterceptor
Type : HttpInterceptorFn
Default value : (req, next) => { const authService = inject(AuthService); const cookieService = inject(CookieHelperService); const tokenService = inject(TokenHelperService); if (req.url.includes('/login') || req.url.includes('/register') || req.url.includes('/send')|| req.url.includes('/validate')) { return next(req); } const activeToken = cookieService.getCookie('ActiveToken'); if (!activeToken) { return next(req); } const token = localStorage.getItem(activeToken!); if (!token) { return next(req); } req = req.clone({ setHeaders: { Authorization: `Bearer ${token}`, }, }); return next(req).pipe( catchError((error) => { if (error.status === 401 && !req.url.includes('/refresh-token')) { return authService.refreshToken().pipe( switchMap((newAccessToken) => { if (!newAccessToken) { return throwError(() => error); } const clonedRequest = req.clone({ setHeaders: { Authorization: `Bearer ${newAccessToken}` } }); return next(clonedRequest); }), catchError((err) => { return throwError(() => err); }) ); } return throwError(() => error); }) ) }
